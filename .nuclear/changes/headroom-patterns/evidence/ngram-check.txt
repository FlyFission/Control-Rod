"""Shared lower-cased word n-grams between new files and source files. Usage: ngram.py N --src F... --new F..."""
import re, sys
args = sys.argv[1:]
n = int(args[0]); i = args.index("--src"); j = args.index("--new")
src, new = args[i+1:j], args[j+1:]
W = re.compile(r"[a-z0-9]+")
def grams(path):
    w = W.findall(open(path, encoding="utf-8").read().lower())
    return {" ".join(w[k:k+n]) for k in range(len(w)-n+1)}
pool = set().union(*(grams(p) for p in src))
for p in new:
    shared = sorted(grams(p) & pool)
    print(f"{p}: {len(shared)} shared {n}-grams {shared}")
